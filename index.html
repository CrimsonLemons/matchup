See URL läheb index.html failis CONFIG.API_URL kohale. * * Skript loob Sheeti automaatselt kaks lehte ("Players" ja "Matches"), * kui neid seal veel pole — midagi käsitsi luua ei ole vaja. */ const PLAYERS_SHEET = 'Players'; const MATCHES_SHEET = 'Matches'; function getSheet_(name, headers) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.appendRow(headers); } return sheet; } function playersSheet_() { return getSheet_(PLAYERS_SHEET, ['name', 'level', 'updatedAt']); } function matchesSheet_() { const sheet = getSheet_(MATCHES_SHEET, [ 'id', 'kind', 'date', 'time', 'location', 'note', 'creatorName', 'participants', 'createdAt' ]); // Force date/time columns to stay as plain text, so Google Sheets doesn't // auto-convert "2026-09-25" or "18:00" into its own Date/Time type. sheet.getRange('C:D').setNumberFormat('@'); return sheet; } function readRows_(sheet) { const values = sheet.getDataRange().getValues(); const headers = values[0]; return values.slice(1) .filter(row => row.some(cell => cell !== '')) .map(row => { const obj = {}; headers.forEach((h, i) => { obj[h] = row[i]; }); return obj; }); } function jsonResponse_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { const tz = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone(); const players = readRows_(playersSheet_()); const matches = readRows_(matchesSheet_()).map(m => { let participants = []; try { participants = JSON.parse(m.participants || '[]'); } catch (err) {} let date = m.date; let time = m.time; // Kaitse vanade, enne parandust juba "Date"-tüübiks moondunud ridade vastu. if (Object.prototype.toString.call(date) === '[object Date]') { date = Utilities.formatDate(date, tz, 'yyyy-MM-dd'); } if (Object.prototype.toString.call(time) === '[object Date]') { time = Utilities.formatDate(time, tz, 'HH:mm'); } return Object.assign({}, m, { date: date, time: time, participants: participants }); }); return jsonResponse_({ players: players, matches: matches }); } function doPost(e) { let body; try { body = JSON.parse(e.postData.contents); } catch (err) { return jsonResponse_({ error: 'Vigane päring' }); } try { switch (body.action) { case 'savePlayer': return jsonResponse_(savePlayer_(body)); case 'createMatch': return jsonResponse_(createMatch_(body)); case 'joinMatch': return jsonResponse_(joinMatch_(body)); case 'leaveMatch': return jsonResponse_(leaveMatch_(body)); case 'cancelMatch': return jsonResponse_(cancelMatch_(body)); default: return jsonResponse_({ error: 'Tundmatu tegevus' }); } } catch (err) { return jsonResponse_({ error: String(err) }); } } function savePlayer_(body) { const sheet = playersSheet_(); const data = sheet.getDataRange().getValues(); const name = String(body.name || '').trim(); if (!name) throw new Error('Nimi puudub'); for (let i = 1; i < data.length; i++) { if (data[i][0] === name) { sheet.getRange(i + 1, 2, 1, 2).setValues([[body.level, new Date().toISOString()]]); return { ok: true }; } } sheet.appendRow([name, body.level, new Date().toISOString()]); return { ok: true }; } function createMatch_(body) { const sheet = matchesSheet_(); const id = Utilities.getUuid(); sheet.appendRow([ id, body.kind, body.date, body.time, body.location || '', body.note || '', body.creatorName, JSON.stringify([body.creatorName]), new Date().toISOString() ]); return { ok: true, id: id }; } function findMatchRow_(sheet, id) { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === id) return i + 1; } return -1; } function joinMatch_(body) { const sheet = matchesSheet_(); const row = findMatchRow_(sheet, body.matchId); if (row === -1) throw new Error('Üleskutset ei leitud'); const kind = sheet.getRange(row, 2).getValue(); const total = kind === 'doubles' ? 4 : 2; let participants = []; try { participants = JSON.parse(sheet.getRange(row, 8).getValue() || '[]'); } catch (e) {} if (participants.indexOf(body.name) === -1 && participants.length < total) { participants.push(body.name); sheet.getRange(row, 8).setValue(JSON.stringify(participants)); } return { ok: true }; } function leaveMatch_(body) { const sheet = matchesSheet_(); const row = findMatchRow_(sheet, body.matchId); if (row === -1) throw new Error('Üleskutset ei leitud'); let participants = []; try { participants = JSON.parse(sheet.getRange(row, 8).getValue() || '[]'); } catch (e) {} participants = participants.filter(p => p !== body.name); sheet.getRange(row, 8).setValue(JSON.stringify(participants)); return { ok: true }; } function cancelMatch_(body) { const sheet = matchesSheet_(); const row = findMatchRow_(sheet, body.matchId); if (row === -1) throw new Error('Üleskutset ei leitud'); const creatorName = sheet.getRange(row, 7).getValue(); if (creatorName !== body.requesterName) throw new Error('Ainult looja saab tühistada'); sheet.deleteRow(row); return { ok: true }; }
